using Doctrine;

// dgo5: the fifth-round instrument. CPU only, BelowNormal, 6 workers.
//   dgo5 smoke <arm>                       timing only
//   dgo5 run <arm> <out.tsv> [games] [searchGames]   all cells of the arm, one TSV row per game
//   dgo5 rerun <arm> <out.tsv>             first 20 games of three searcher cells (N4)
//   dgo5 check <a.tsv> <b.tsv>             rows of b equal the same rows of a (N4)
//   dgo5 oracle <arm> <games>              N3: the rollout world predicts the real round exactly
//   dgo5 report <a.tsv> [b.tsv ...]        the tables and targets
//   dgo5 rot <arm> <out.txt> [seeds]       the three rotations on the arm's game
static class P
{
    static readonly Stopwatch Clock = Stopwatch.StartNew();
    static void Log(string s) => Console.WriteLine($"[{Clock.Elapsed.TotalMinutes:F1} min] {s}");
    public const int BaseSeed = 10_100_000;
    public const int EnemyPct = 25;

    public static readonly string[] ArmNames = ["G0", "G1", "G2", "G3"];

    static (DoctrineDomain Rules, TracerBattleOptions Opt) ArmConfig(string arm)
    {
        var d = new DoctrineDomain { MaxRounds = 8, Offers = OfferRules.Default, EnemyBaseGold = 14 * EnemyPct / 100, EnemyGoldPerRound = 4 * EnemyPct / 100 };
        return arm switch
        {
            "G0" => (new DoctrineDomain { MaxRounds = 8, Offers = OfferRules.Default, EnemyBaseGold = d.EnemyBaseGold, EnemyGoldPerRound = d.EnemyGoldPerRound, Siege = false }, new TracerBattleOptions { Set = false, SiegeClock = false }),
            "G1" => (d, new TracerBattleOptions()),
            "G2" => (new DoctrineDomain { MaxRounds = 8, Offers = OfferRules.Default, EnemyBaseGold = d.EnemyBaseGold, EnemyGoldPerRound = d.EnemyGoldPerRound, Comeback = true }, new TracerBattleOptions()),
            "G3" => (new DoctrineDomain { MaxRounds = 8, Offers = OfferRules.Default, EnemyBaseGold = d.EnemyBaseGold, EnemyGoldPerRound = d.EnemyGoldPerRound, FlatSpoils = true }, new TracerBattleOptions()),
            _ => throw new ArgumentException("arm " + arm),
        };
    }

    sealed class RecGeneral : IGeneral
    {
        public int ExamplesSeen => 0;
        public ValueTask<GeneralOutput> SuggestAsync(GeneralInput input, CancellationToken ct) => ValueTask.FromResult(new GeneralOutput(DoctrineSet.Empty, 0));
        public ValueTask LearnAsync(GeneralExample example, CancellationToken ct) => ValueTask.CompletedTask;
        public ValueTask LearnOutcomeAsync(GeneralOutcome outcome, CancellationToken ct) => ValueTask.CompletedTask;
    }

    sealed class Worker(string scratch, string arm)
    {
        public DoctrineGame Game = null!; public SwitchPlayer Sw = null!; public RollWorld Roll = null!; public int Retries;
        public void Init()
        {
            var (rules, opt) = ArmConfig(arm);
            DoctrineGame Make(IPlayerInput p, PleromaArmyService svc)
            {
                Directory.CreateDirectory(scratch);
                var world = new Tracer.PathfindingWorld(tunerStatePath: Path.Combine(scratch, Guid.NewGuid().ToString("N")));
                return DoctrineBuilder.New().WithDefaults(r => r.WithArmy(svc).WithTracer(world).WithBattle(ctx => new TracerBattleEngine(ctx.Tracer!, opt)).WithGeneral(new RecGeneral()))
                    .WithPlayer(p).WithRules(rules).WithSingleThread(true).Build();
            }
            Sw = new SwitchPlayer();
            Game = Make(Sw, new PleromaArmyService());
            var rsvc = new PleromaArmyService();
            Roll = new RollWorld(Make(new SwitchPlayer(), rsvc), rsvc, rules);
        }
    }

    static T[] Run<T>(string arm, int n, Func<int, Worker, T> unit, string what)
    {
        var res = new T[n]; int next = -1, done = 0, last = -1;
        var threads = new List<Thread>(); Exception? fail = null;
        string root = Path.Combine(Path.GetTempPath(), "dgo5-tuner", Guid.NewGuid().ToString("N"));
        for (int w = 0; w < Math.Min(6, n); w++)
        {
            int wi = w;
            var th = new Thread(() =>
            {
                try
                {
                    var wk = new Worker(Path.Combine(root, wi.ToString()), arm); wk.Init();
                    while (true)
                    {
                        int i = Interlocked.Increment(ref next);
                        if (i >= n) break;
                        res[i] = unit(i, wk);
                        int d = Interlocked.Increment(ref done), pct = d * 20 / n;
                        if (pct != last) { last = pct; Log($"{what} {d}/{n}"); }
                    }
                }
                catch (Exception e) { fail = e; Interlocked.Exchange(ref next, n); }
            }, 16 * 1024 * 1024) { IsBackground = true };
            th.Start(); threads.Add(th);
        }
        foreach (var t in threads) t.Join();
        if (fail is not null) throw new AggregateException(fail);
        return res;
    }

    public const int Clones = 6;
    static readonly string[] Scripts = Styles.Six.Select(s => s.Name).ToArray();

    static List<(string Policy, bool Search)> Cells(string arm)
    {
        var l = new List<(string, bool)>();
        if (arm == "G3") { for (int c = 0; c < Clones; c++) l.Add(("RANDOM" + c, false)); return l; }
        foreach (var s in Scripts) l.Add((s, false));
        for (int c = 0; c < Clones; c++) l.Add(("RANDOM" + c, false));
        l.Add(("SANDBAG", false));
        l.Add(("SEARCH-NOVICE", true)); l.Add(("SEARCH-EXPERT", true)); l.Add(("SHUFFLED-EXPERT", true));
        return l;
    }

    static GameLog PlayGame(Worker wk, string policy, int seed, bool oracle = false)
    {
        for (int attempt = 0; ; attempt++)
        {
            IPlayerInput inner; Func<GameLog> log;
            if (oracle) { var op = new OraclePlayer(wk.Roll, 1 + seed % 3, seed); inner = op; log = () => op.Log; }
            else if (policy.StartsWith("RANDOM")) { var rp = new RandomPlayer(int.Parse(policy[6..]) + 1, seed); inner = rp; log = () => rp.Log; }
            else if (policy == "SANDBAG") { var sp = new SandbagPlayer(1, seed); inner = sp; log = () => sp.Log; }
            else if (policy.StartsWith("SEARCH") || policy.StartsWith("SHUFFLED"))
            {
                var sp = new SearchPlayer(wk.Roll, policy.EndsWith("NOVICE") ? Levels.Novice : Levels.Expert, seed, policy.StartsWith("SHUFFLED")); inner = sp; log = () => sp.Log;
            }
            else { var sp = new StylePlayer(Styles.Six.First(s => s.Name == policy), 17, seed); inner = sp; log = () => sp.Log; }
            wk.Sw.Inner = inner;
            try
            {
                var run = wk.Game.Context.Round.RunAsync(wk.Game.NewGame(seed)).AsTask().GetAwaiter().GetResult();
                if (run is EvalApp.PipelineResult<RoundState>.Failure f)
                {
                    if ((f.Exception is null || f.Exception is NullReferenceException || (f.Exception is InvalidOperationException ioe && ioe.Message.Contains("no exception"))) && attempt < 8) { wk.Retries++; continue; }
                    throw new InvalidOperationException("pipeline failure: " + (f.Exception?.ToString() ?? f.Message ?? "no detail"));
                }
                return log();
            }
            catch (NullReferenceException) when (attempt < 8) { wk.Retries++; }
        }
    }

    const string Header = "arm\tpolicy\tseed\tscores\tscriptRounds\tscriptHits\trefused\tillegal\trounds\tsiege\tclock\tmuster\toracleChecked\toracleWrong\n";
    static string RowOf(string arm, string pol, int seed, GameLog l)
        => $"{arm}\t{pol}\t{seed}\t{string.Join(",", l.Scores.Select(x => x.ToString("0.0", CultureInfo.InvariantCulture)))}\t{l.ScriptRounds}\t{l.ScriptHits}\t{l.Refused}\t{l.Illegal}\t{l.Rounds}\t{l.SiegeRounds}\t{l.ClockRounds}\t{l.Muster}\t{l.OracleChecked}\t{l.OracleWrong}\n";

    static void RunArm(string arm, string path, int games, int searchGames, int first = 0, string[]? only = null)
    {
        var units = new List<(string Policy, int Seed)>();
        foreach (var (pol, search) in Cells(arm))
        {
            if (only is not null && !only.Contains(pol)) continue;
            int n = first > 0 ? first : search ? searchGames : games;
            for (int g = 0; g < n; g++) units.Add((pol, BaseSeed + g));
        }
        // longest jobs first so the pool does not end waiting on a few slow searcher games
        var order = units.Select((u, i) => (u, i)).OrderByDescending(x => x.u.Policy.StartsWith("SEARCH") || x.u.Policy.StartsWith("SHUFFLED") ? (x.u.Policy.EndsWith("EXPERT") ? 2 : 1) : 0).Select(x => x.u).ToList();
        var rows = Run(arm, order.Count, (i, wk) => (order[i], PlayGame(wk, order[i].Policy, order[i].Seed)), arm);
        var sb = new StringBuilder(Header);
        foreach (var (u, l) in rows.OrderBy(r => r.Item1.Policy, StringComparer.Ordinal).ThenBy(r => r.Item1.Seed)) sb.Append(RowOf(arm, u.Policy, u.Seed, l));
        File.WriteAllText(path, sb.ToString());
    }

    sealed record Row(string Arm, string Policy, int Seed, double[] Scores, int Script, int Hits, int Refused, int Illegal, int Rounds, int Siege, int Clock, int Muster, int OChecked, int OWrong)
    {
        public double Mean => Scores.Average();
        public int Wins => Scores.Count(x => x >= 1.0);
        public double Rest => Scores.Skip(1).Average();
    }

    static List<Row> Load(string path) => File.ReadLines(path).Skip(1).Select(l => l.Split('\t')).Select(f => new Row(f[0], f[1], int.Parse(f[2]),
        f[3].Split(',').Select(x => double.Parse(x, CultureInfo.InvariantCulture)).ToArray(), int.Parse(f[4]), int.Parse(f[5]), int.Parse(f[6]), int.Parse(f[7]), int.Parse(f[8]),
        int.Parse(f[9]), int.Parse(f[10]), int.Parse(f[11]), int.Parse(f[12]), int.Parse(f[13]))).ToList();

    static double Sd(IEnumerable<double> x) { var a = x.ToArray(); if (a.Length < 2) return 0; double m = a.Average(); return Math.Sqrt(a.Sum(v => (v - m) * (v - m)) / (a.Length - 1)); }

    static (double Mean, double Se, int N) Cell(IEnumerable<Row> rows) { var m = rows.Select(r => r.Mean).ToArray(); return (m.Average(), Sd(m) / Math.Sqrt(m.Length), m.Length); }

    static (double Mom, double Se, int N1, int N0) Momentum(IEnumerable<Row> rows)
    {
        var win = rows.Where(r => r.Scores[0] >= 1.0).Select(r => r.Rest).ToArray(); var loss = rows.Where(r => r.Scores[0] <= 0.0).Select(r => r.Rest).ToArray();
        if (win.Length < 2 || loss.Length < 2) return (double.NaN, double.NaN, win.Length, loss.Length);
        double se = Math.Sqrt(Math.Pow(Sd(win), 2) / win.Length + Math.Pow(Sd(loss), 2) / loss.Length);
        return (win.Average() - loss.Average(), se, win.Length, loss.Length);
    }

    static void Report(string[] paths)
    {
        var all = paths.SelectMany(Load).ToList();
        var sb = new StringBuilder();
        void W(string s) { sb.AppendLine(s); Console.WriteLine(s); }
        W("mean = mean round score over 8 rounds (win 1, draw .5, loss 0; unplayed rounds of a dead army 0); SE over games");
        var cells = new Dictionary<(string Arm, string Pol), (double Mean, double Se, int N)>();
        var moms = new Dictionary<string, (double Mom, double Se, int N1, int N0)>();
        foreach (var arm in ArmNames.Where(a => all.Any(r => r.Arm == a)))
        {
            var rows = all.Where(r => r.Arm == arm).ToList();
            W($"\n== {arm}");
            W($"{"policy",-16} {"games",5} {"mean",6} {"SE",6} {"solved",6} {"script",6} {"siege/g",7} {"clock/g",7} {"muster/g",8} {"refused",7} {"illegal",7}");
            foreach (var g in rows.GroupBy(r => r.Policy).OrderBy(g => g.Key, StringComparer.Ordinal))
            {
                var c = Cell(g); cells[(arm, g.Key)] = c;
                double script = g.Sum(r => r.Script) == 0 ? double.NaN : (double)g.Sum(r => r.Hits) / g.Sum(r => r.Script);
                W($"{g.Key,-16} {c.N,5} {c.Mean,6:F3} {c.Se,6:F3} {g.Average(r => r.Wins >= 6 ? 1.0 : 0.0),6:F3} {script,6:F2} {g.Average(r => r.Siege),7:F2} {g.Average(r => r.Clock),7:F2} {g.Average(r => r.Muster),8:F1} {g.Sum(r => r.Refused),7} {g.Sum(r => r.Illegal),7}");
            }
            var rnd = rows.Where(r => r.Policy.StartsWith("RANDOM")).ToList();
            if (rnd.Count > 0)
            {
                var rc = Cell(rnd); cells[(arm, "RANDOM*")] = rc;
                var clones = rnd.GroupBy(r => r.Policy).Select(g => g.Average(r => r.Mean)).ToArray();
                W($"RANDOM pooled {rc.Mean:F3} (SE {rc.Se:F3}, {rc.N} games); clone range {clones.Max() - clones.Min():F3}; analytic floor 2.53 SE_clone {2.53 * Sd(rnd.Select(r => r.Mean)) / Math.Sqrt(rnd.Count / Math.Max(1, clones.Length)):F3}");
                moms[arm] = Momentum(rows.Where(r => r.Policy != "SANDBAG"));
            }
            var sc = Scripts.Where(s => cells.ContainsKey((arm, s))).OrderByDescending(s => cells[(arm, s)].Mean).ToList();
            if (sc.Count >= 2)
            {
                double gap = cells[(arm, sc[0])].Mean - cells[(arm, sc[1])].Mean, sed = Math.Sqrt(Math.Pow(cells[(arm, sc[0])].Se, 2) + Math.Pow(cells[(arm, sc[1])].Se, 2));
                W($"scripts: spread {cells[(arm, sc[0])].Mean - cells[(arm, sc.Last())].Mean:F3}; best {sc[0]} {cells[(arm, sc[0])].Mean:F3}, second {sc[1]} {cells[(arm, sc[1])].Mean:F3}, gap {gap:F3} ({gap / sed:F1} SE_d)  TA4 {(gap < 0.05 || gap < 2.5 * sed ? "MET" : "FAIL")}");
            }
            bool has(string p) => cells.ContainsKey((arm, p));
            if (has("SEARCH-EXPERT") && has("RANDOM*"))
            {
                double D(string a, string b, out double sed) { sed = Math.Sqrt(Math.Pow(cells[(arm, a)].Se, 2) + Math.Pow(cells[(arm, b)].Se, 2)); return cells[(arm, a)].Mean - cells[(arm, b)].Mean; }
                double d1 = D("SEARCH-EXPERT", "RANDOM*", out var s1);
                W($"TA1 expert - random {d1:+0.000;-0.000} ({d1 / s1:F1} SE_d)  {(d1 >= 0.05 && d1 >= 3 * s1 ? "MET" : "FAIL")}");
                if (has("SEARCH-NOVICE")) { double d2 = D("SEARCH-EXPERT", "SEARCH-NOVICE", out var s2); W($"TA2 expert - novice {d2:+0.000;-0.000} ({d2 / s2:F1} SE_d)  {(d2 >= 0.03 && d2 >= 2 * s2 ? "MET" : "FAIL")};  novice - random {D("SEARCH-NOVICE", "RANDOM*", out _):+0.000;-0.000}"); }
                if (has("SHUFFLED-EXPERT")) { double d3 = D("SEARCH-EXPERT", "SHUFFLED-EXPERT", out var s3); W($"TA3 expert - shuffled {d3:+0.000;-0.000} ({d3 / s3:F1} SE_d)  {(d3 >= 0.04 && d3 >= 2.5 * s3 ? "MET" : "FAIL")};  shuffled - random {D("SHUFFLED-EXPERT", "RANDOM*", out _):+0.000;-0.000}"); }
                if (has("SANDBAG")) W($"SANDBAG {cells[(arm, "SANDBAG")].Mean:F3} against RANDOM0 {cells[(arm, "RANDOM0")].Mean:F3}");
            }
            var mm = moms.GetValueOrDefault(arm);
            if (!double.IsNaN(mm.Mom)) W($"MOMENTUM (rounds 1-7 after a round-0 win minus after a loss) {mm.Mom:+0.000;-0.000} (SE {mm.Se:F3}; {mm.N1} wins, {mm.N0} losses)");
        }
        if (moms.ContainsKey("G1") && moms.ContainsKey("G3"))
        {
            var g1 = moms["G1"]; var g3 = moms["G3"];
            double floorGap = g1.Mom - g3.Mom, se13 = Math.Sqrt(g1.Se * g1.Se + g3.Se * g3.Se);
            W($"\nSNOWBALL in G1 over the flat-spoils null: {floorGap:+0.000;-0.000} ({floorGap / se13:F1} SE)");
            if (moms.ContainsKey("G2")) { double rem = (g1.Mom - moms["G2"].Mom) / floorGap; W($"MOMENTUM REMOVED by the comeback (G1 -> G2): {rem:F2}"); }
        }
        File.WriteAllText(Path.ChangeExtension(paths[0], ".report.txt"), sb.ToString());
    }

    static void Check(string a, string b)
    {
        var A = Load(a).ToDictionary(r => (r.Arm, r.Policy, r.Seed)); int bad = 0, n = 0;
        foreach (var r in Load(b)) { n++; if (!A.TryGetValue((r.Arm, r.Policy, r.Seed), out var x) || !x.Scores.SequenceEqual(r.Scores)) bad++; }
        Log($"N4 rerun: {n} rows compared, {bad} differ");
    }

    static void Rot(string arm, string path, int seeds)
    {
        var docs = new (string Name, int[] Tokens)[]
        {
            ("ADV/HOLD/FLANK", [DoctrineVocabulary.Field(FieldOrderKind.Advance, 0), DoctrineVocabulary.Field(FieldOrderKind.Hold, 1), DoctrineVocabulary.Field(FieldOrderKind.Flank, 2)]),
            ("HOLD/ADV/FALLBACK", [DoctrineVocabulary.Field(FieldOrderKind.Hold, 0), DoctrineVocabulary.Field(FieldOrderKind.Advance, 1), DoctrineVocabulary.Field(FieldOrderKind.FallBack, 2)]),
        };
        var units = new List<(int Doc, int Seed, int Variant)>();
        for (int d = 0; d < docs.Length; d++) for (int s = 0; s < seeds; s++) for (int v = 0; v < 6; v++) units.Add((d, 9_950_000 + s, v));
        var svc = new PleromaArmyService();
        string[] arenas = ["open-field", "chokepoint", "ridge", "falling-gate"];
        var (rules, _) = ArmConfig(arm);
        var res = Run(arm, units.Count, (i, wk) =>
        {
            var u = units[i];
            for (int attempt = 0; ; attempt++)
                try
                {
                    IArmy Raise(Side side, int seed, int salt)
                    {
                        var a = svc.Create(side, seed);
                        a.ApplySpoilsAsync(new Spoils(24, 0), default).AsTask().GetAwaiter().GetResult();
                        var offer = SeededOffers.Instance.Units(seed, salt, a.Read().Gold, OfferRules.Default);
                        var rng = new GameRng(GameRng.Mix(seed, 31 + salt)); var left = offer.Packs.ToList(); int gold = a.Read().Gold; var picks = new List<DraftPick>();
                        for (int k = 0; k < 3; k++) { var can = left.Where(p => p.Cost <= gold).ToList(); if (can.Count == 0) break; var p = can[rng.Next(can.Count)]; picks.Add(UnitOffer.Pick(p)); gold -= p.Cost; left.Remove(p); }
                        a.DraftAsync(new DraftRequest(0, seed, picks), default).AsTask().GetAwaiter().GetResult();
                        return a;
                    }
                    var player = Raise(Side.Player, u.Seed, 0); var enemy = Raise(Side.Enemy, u.Seed, 1);
                    int rot = u.Variant < 3 ? u.Variant : 0;
                    if (rot > 0) player.ApplyArmyOrdersAsync(DoctrineSet.Of(DoctrineVocabulary.Army(ArmyOrderKind.Rotate, rot)), default).AsTask().GetAwaiter().GetResult();
                    var doc = DoctrineSet.Of(docs[u.Doc].Tokens); var edoc = DoctrineSet.Of(DoctrineVocabulary.Field(FieldOrderKind.Advance, 0), DoctrineVocabulary.Field(FieldOrderKind.Advance, 1), DoctrineVocabulary.Field(FieldOrderKind.Advance, 2));
                    int battleSeed = u.Seed * 7 + 3 + (u.Variant >= 3 ? u.Variant - 2 : 0);
                    var aid = arenas[u.Seed % 4]; var info = wk.Game.Context.Battle.Arenas.First(a => a.Id == aid);
                    Side? bes = rules.Siege && info.IsSiege ? SiegeRules.BesiegerOf(0, rules.SwapSiegeRoles) : null;
                    var setup = new BattleSetup(aid, battleSeed, 120,
                        new BattleSide(Side.Player, player.Read().Cohorts, DoctrineVocabulary.FieldOrders(doc), DoctrineVocabulary.Trigger(doc)),
                        new BattleSide(Side.Enemy, enemy.Read().Cohorts, DoctrineVocabulary.FieldOrders(edoc), DoctrineVocabulary.Trigger(edoc)), bes);
                    var run = wk.Game.Context.BattleRun.RunAsync(new BattleState(setup)).AsTask().GetAwaiter().GetResult();
                    if (run is EvalApp.PipelineResult<BattleState>.Failure f) throw f.Exception ?? new InvalidOperationException("no exception");
                    var o = run.GetData().Result!.Outcome;
                    return o == BattleStatus.AttackerWon ? 1.0 : o == BattleStatus.Draw ? 0.5 : 0.0;
                }
                catch (Exception) when (attempt < 8) { wk.Retries++; }
        }, "rot " + arm);
        var sb = new StringBuilder();
        void W(string s) { sb.AppendLine(s); Console.WriteLine(s); }
        for (int d = 0; d < docs.Length; d++)
        {
            double[,] sc = new double[seeds, 6];
            for (int i = 0; i < units.Count; i++) if (units[i].Doc == d) sc[units[i].Seed - 9_950_000, units[i].Variant] = res[i];
            double[] mean = new double[6];
            for (int v = 0; v < 6; v++) { double s = 0; for (int k = 0; k < seeds; k++) s += sc[k, v]; mean[v] = s / seeds; }
            double Oracle(int a) { double g = 0; for (int k = 0; k < seeds; k++) g += Math.Max(sc[k, a], Math.Max(sc[k, a + 1], sc[k, a + 2])) - (sc[k, a] + sc[k, a + 1] + sc[k, a + 2]) / 3; return g / seeds; }
            double Disagree(int a) { int n = 0; for (int k = 0; k < seeds; k++) if (sc[k, a] != sc[k, a + 1] || sc[k, a] != sc[k, a + 2]) n++; return n / (double)seeds; }
            W($"{arm} {docs[d].Name}: rotation 0/1/2 mean win {mean[0]:F3} {mean[1]:F3} {mean[2]:F3} (spread {mean[0..3].Max() - mean[0..3].Min():F3}); null {mean[3]:F3} {mean[4]:F3} {mean[5]:F3} (spread {mean[3..6].Max() - mean[3..6].Min():F3})");
            W($"{arm} {docs[d].Name}: oracle gain rotations {Oracle(0):F3}, null {Oracle(3):F3}, excess {Oracle(0) - Oracle(3):F3}; outcomes differing across the three: rotations {Disagree(0):F2}, null {Disagree(3):F2}");
        }
        File.WriteAllText(path, sb.ToString());
    }

    static void Main(string[] args)
    {
        try { Process.GetCurrentProcess().PriorityClass = ProcessPriorityClass.BelowNormal; } catch { }
        switch (args[0])
        {
            case "smoke":
                {
                    var sw = Stopwatch.StartNew(); string arm = args[1];
                    foreach (var pol in new[] { "MASS", "RANDOM0", "SEARCH-NOVICE", "SEARCH-EXPERT", "SHUFFLED-EXPERT" })
                    {
                        sw.Restart();
                        var r = Run(arm, 6, (i, wk) => PlayGame(wk, pol, 1 + i), "smoke " + pol);
                        Log($"smoke {pol}: 6 games {sw.Elapsed.TotalSeconds:F1} s wall; mean {r.Average(x => x.Scores.Average()):F2}; refused {r.Sum(x => x.Refused)} illegal {r.Sum(x => x.Illegal)}");
                    }
                    break;
                }
            case "run": RunArm(args[1], args[2], args.Length > 3 ? int.Parse(args[3]) : 200, args.Length > 4 ? int.Parse(args[4]) : 100); Log("done"); break;
            case "rerun": RunArm(args[1], args[2], 0, 0, 20, ["SEARCH-NOVICE", "SEARCH-EXPERT", "SHUFFLED-EXPERT"]); Log("done"); break;
            case "check": Check(args[1], args[2]); break;
            case "oracle":
                {
                    string arm = args[1]; int games = int.Parse(args[2]);
                    var r = Run(arm, games, (i, wk) => PlayGame(wk, "ORACLE", BaseSeed + 50_000 + i, oracle: true), "oracle");
                    Log($"N3 {arm}: {r.Sum(x => x.OracleChecked)} rounds predicted from the rollout world with the true enemy orders and battle seed; {r.Sum(x => x.OracleWrong)} wrong");
                    break;
                }
            case "report": Report(args.Skip(1).ToArray()); break;
            case "rot": Rot(args[1], args[2], args.Length > 3 ? int.Parse(args[3]) : 240); break;
        }
    }
}

/// <summary>N3: a chance player that, at its choice, imagines the round with the TRUE enemy orders and the TRUE battle seed and records what the rollout world predicts.</summary>
public sealed class OraclePlayer(RollWorld world, int salt, int seed) : IPlayerInput
{
    readonly RandomPlayer _p = new(salt, seed);
    readonly Dictionary<int, double> _pred = new();
    public GameLog Log => _p.Log;
    public ValueTask<IReadOnlyList<DraftPick>> DraftAsync(ArmyReadModel a, int r, CancellationToken ct) => _p.DraftAsync(a, r, ct);
    public ValueTask<IReadOnlyList<DraftPick>> DraftAsync(ArmyReadModel a, UnitOffer o, int r, CancellationToken ct) => _p.DraftAsync(a, o, r, ct);
    public ValueTask<PlayerChoice> ChooseAsync(ArmyReadModel a, GeneralOutput g, int r, CancellationToken ct) => _p.ChooseAsync(a, g, r, ct);
    public async ValueTask<PlayerChoice> ChooseAsync(ArmyReadModel a, GeneralOutput g, OrderOffer o, int r, CancellationToken ct)
    {
        var choice = await _p.ChooseAsync(a, g, o, r, ct);
        // the pipeline drops a field or trigger token the start mask does not allow
        int Keep(int t) => t != 0 && DoctrineVocabulary.IsValid(t) && !(DoctrineVocabulary.Decode(t).Family is OrderFamily.Field or OrderFamily.Trigger && !a.Legal.Allows(t)) ? t : 0;
        var c = choice.Doctrine; var doc = new DoctrineSet(Keep(c.S0), Keep(c.S1), Keep(c.S2), Keep(c.S3));
        var me = world.Rebuild(a);
        await me.ApplyArmyOrdersAsync(doc, ct);
        var enemy = world.Enemy(seed, r);
        var edoc = ScriptedOpponent.DoctrineFor(r, seed, enemy.Read().Legal);   // the TRUE enemy orders
        await enemy.ApplyArmyOrdersAsync(edoc, ct);
        _pred[r] = world.Fight(seed, r, me, doc, enemy, edoc, (int)GameRng.Mix(seed, 4000 + r)).Result;   // the TRUE battle seed
        return choice;
    }
    public ValueTask RoundClosedAsync(RoundRecord rec, ArmyReadModel after, CancellationToken ct)
    {
        double actual = rec.Result.Outcome == BattleStatus.AttackerWon ? 1 : rec.Result.Outcome == BattleStatus.Draw ? 0.5 : 0;
        Log.OracleChecked++; if (!_pred.TryGetValue(rec.Round, out var p) || p != actual) Log.OracleWrong++;
        return _p.RoundClosedAsync(rec, after, ct);
    }
}

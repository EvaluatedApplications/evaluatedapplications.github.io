// Fifth-round instrument, part A (Doctrine\todo\difficulty-2026-10-19.md, "FIFTH ROUND"). A player that SEARCHES the offer with short rollouts on the real game.

public sealed record Level(string Name, int DraftWidth, int DraftRolls, int OrderWidth, int OrderRolls);

public static class Levels
{
    public static readonly Level Novice = new("NOVICE", 3, 1, 3, 1);
    public static readonly Level Expert = new("EXPERT", 8, 2, 16, 2);
}

/// <summary>A second game per worker: the same sections and switches as the real one, used only to fight imagined rounds. It never touches the real armies.</summary>
public sealed class RollWorld(DoctrineGame game, PleromaArmyService svc, DoctrineDomain rules)
{
    public int Battles;
    public DoctrineDomain Rules => rules;

    /// <summary>The player's army rebuilt from its read (before any order of the round: no aura stands): the army code of that read, parsed by the army section.</summary>
    public IArmy Rebuild(ArmyReadModel a)
    {
        int flip = a.Cohorts.Count == 0 ? 0 : (int)a.Cohorts[0].Traits ^ (int)Roster.TraitsOf(a.Cohorts[0].Key);
        var code = $"DA2.{(int)a.Side}.{a.Round}.{a.Gold}.{flip}.." + string.Join(";", a.Cohorts.Select(c => $"{c.Key}:{c.Count}:{c.Slot}:{c.Veterancy}"));
        if (!svc.TryParse(code, out var c2, out var err) || c2 is null) throw new InvalidOperationException("rebuild failed: " + err + " code " + code);
        return c2;
    }

    public IArmy Clone(IArmy a)
    {
        if (!svc.TryParse(a.ToCode(), out var c, out var err) || c is null) throw new InvalidOperationException("clone failed: " + err);
        return c;
    }

    /// <summary>The enemy as the game raises it for (seed, round): fresh levy, scripted draft. Its orders are NOT applied.</summary>
    public IArmy Enemy(int seed, int round)
    {
        var e = svc.Create(Side.Enemy, (int)GameRng.Mix(seed, round));
        int gold = rules.EnemyBaseGold + rules.EnemyGoldPerRound * round;
        e.DraftAsync(new DraftRequest(round, seed, ScriptedOpponent.Picks(round, seed, gold)), default).AsTask().GetAwaiter().GetResult();
        return e;
    }

    public ArenaInfo ArenaOf(int seed, int round)
    {
        var arenas = game.Context.Battle.Arenas;
        return arenas[(int)(GameRng.Mix(seed, 3000 + round) % (ulong)arenas.Count)];
    }

    public (double Result, double Margin) Fight(int seed, int round, IArmy player, DoctrineSet pdoc, IArmy enemy, DoctrineSet edoc, int battleSeed)
    {
        var arena = ArenaOf(seed, round);
        Side? bes = rules.Siege && arena.IsSiege ? SiegeRules.BesiegerOf(round, rules.SwapSiegeRoles) : null;
        var pc = player.Read().Cohorts; var ec = enemy.Read().Cohorts;
        var setup = new BattleSetup(arena.Id, battleSeed, rules.TickLimit,
            new BattleSide(Side.Player, pc, DoctrineVocabulary.FieldOrders(pdoc), DoctrineVocabulary.Trigger(pdoc)),
            new BattleSide(Side.Enemy, ec, DoctrineVocabulary.FieldOrders(edoc), DoctrineVocabulary.Trigger(edoc)), bes);
        for (int attempt = 0; ; attempt++)
            try
            {
                var run = game.Context.BattleRun.RunAsync(new BattleState(setup)).AsTask().GetAwaiter().GetResult();
                if (run is EvalApp.PipelineResult<BattleState>.Failure f) throw f.Exception ?? new InvalidOperationException("no exception");
                var r = run.GetData().Result!;
                Interlocked.Increment(ref Battles);
                int own0 = pc.Sum(c => c.Count), en0 = ec.Sum(c => c.Count);
                int ownLeft = own0 - r.AttackerLosses.Sum(l => l.Lost), enLeft = en0 - r.DefenderLosses.Sum(l => l.Lost);
                double res = r.Outcome == BattleStatus.AttackerWon ? 1 : r.Outcome == BattleStatus.Draw ? 0.5 : 0;
                return (res, (ownLeft - enLeft) / (double)Math.Max(1, own0 + en0));
            }
            catch (Exception) when (attempt < 8) { }
    }
}

public sealed class SearchPlayer(RollWorld world, Level level, int seed, bool shuffled, ulong salt = 99) : IPlayerInput
{
    readonly GameRng _rng = new(GameRng.Mix((int)(salt & 0x7fffffff), seed));
    public GameLog Log = new();
    static readonly DoctrineSet Adv = DoctrineSet.Of(DoctrineVocabulary.Field(FieldOrderKind.Advance, 0), DoctrineVocabulary.Field(FieldOrderKind.Advance, 1), DoctrineVocabulary.Field(FieldOrderKind.Advance, 2));

    public ValueTask<IReadOnlyList<DraftPick>> DraftAsync(ArmyReadModel a, int r, CancellationToken ct) => ValueTask.FromResult<IReadOnlyList<DraftPick>>([]);

    static List<List<T>> Subsets<T>(IReadOnlyList<T> items, int maxSize, Func<List<T>, bool> ok)
    {
        var all = new List<List<T>>();
        for (int mask = 1; mask < (1 << items.Count); mask++)
        {
            var s = new List<T>();
            for (int i = 0; i < items.Count; i++) if ((mask >> i & 1) != 0) s.Add(items[i]);
            if (s.Count <= maxSize && ok(s)) all.Add(s);
        }
        return all;
    }

    void Shuffle<T>(IList<T> a) { for (int i = a.Count - 1; i > 0; i--) { int j = _rng.Next(i + 1); (a[i], a[j]) = (a[j], a[i]); } }

    /// <summary>Pick the best of the candidates by rollouts (or, SHUFFLED, the best of the scores permuted among the candidates).</summary>
    int Choose(int n, Func<int, double> score)
    {
        var scores = new double[n];
        for (int i = 0; i < n; i++) scores[i] = score(i);
        if (shuffled) { var idx = Enumerable.Range(0, n).ToList(); Shuffle(idx); scores = idx.Select(i => scores[i]).ToArray(); }
        int best = 0;
        for (int i = 1; i < n; i++) if (scores[i] > scores[best]) best = i;
        return best;
    }

    double Score(int round, int cand, IArmy player, DoctrineSet doc, int rolls, bool orders)
    {
        double sum = 0;
        for (int k = 0; k < rolls; k++)
        {
            var me = world.Clone(player);                                    // the candidate's own copy of the army (a rollout never touches the real one)
            int rollSeed = (int)GameRng.Mix(seed, 70000 + round * 1000 + cand * 10 + k + (orders ? 500 : 0));
            if (orders) me.ApplyArmyOrdersAsync(doc, default).AsTask().GetAwaiter().GetResult();
            var enemy = world.Enemy(seed, round);
            var edoc = ScriptedOpponent.DoctrineFor(round, rollSeed, enemy.Read().Legal);   // the enemy's rule, not its roll
            enemy.ApplyArmyOrdersAsync(edoc, default).AsTask().GetAwaiter().GetResult();
            var (res, margin) = world.Fight(seed, round, me, doc, enemy, edoc, (int)GameRng.Mix(rollSeed, 1));
            sum += res + 0.25 * margin;
        }
        return sum / rolls;
    }

    public ValueTask<IReadOnlyList<DraftPick>> DraftAsync(ArmyReadModel a, UnitOffer offer, int round, CancellationToken ct)
    {
        if (!offer.Active || offer.Packs.Count == 0) return ValueTask.FromResult<IReadOnlyList<DraftPick>>([]);
        var packs = offer.Packs.ToList();
        var subsets = Subsets(packs, packs.Count, s => s.Sum(p => p.Cost) <= a.Gold);
        Shuffle(subsets);
        var cands = new List<List<UnitPack>> { new() };
        cands.AddRange(subsets.Take(level.DraftWidth - 1));
        var real = world.Rebuild(a);
        int best = Choose(cands.Count, i =>
        {
            var test = world.Clone(real);
            test.DraftAsync(new DraftRequest(round, seed, cands[i].Select(UnitOffer.Pick).ToList()), default).AsTask().GetAwaiter().GetResult();
            return Score(round, i, test, Adv, level.DraftRolls, orders: false);
        });
        return ValueTask.FromResult<IReadOnlyList<DraftPick>>(cands[best].Select(UnitOffer.Pick).ToList());
    }

    public ValueTask<PlayerChoice> ChooseAsync(ArmyReadModel a, GeneralOutput g, int round, CancellationToken ct) => ChooseAsync(a, g, OrderOffer.Open(), round, ct);

    public ValueTask<PlayerChoice> ChooseAsync(ArmyReadModel a, GeneralOutput g, OrderOffer offer, int round, CancellationToken ct)
    {
        var tokens = (offer.Active ? offer.Tokens : []).Where(t => a.Legal.Allows(t)).ToList();
        var subsets = Subsets(tokens, 4, _ => true);
        Shuffle(subsets);
        var cands = new List<List<int>> { new() };
        cands.AddRange(subsets.Take(level.OrderWidth - 1));
        var real = world.Rebuild(a);
        int best = Choose(cands.Count, i => Score(round, i, real, DoctrineSet.Of(cands[i].ToArray()), level.OrderRolls, orders: true));
        return ValueTask.FromResult(new PlayerChoice(DoctrineSet.Of(cands[best].ToArray()), true));
    }

    public ValueTask RoundClosedAsync(RoundRecord rec, ArmyReadModel after, CancellationToken ct) { Log.Close(rec); return ValueTask.CompletedTask; }
}

/// <summary>Throws round 0 away: no draft, no orders; then plays like its clone of the chance player.</summary>
public sealed class SandbagPlayer(int salt, int seed) : IPlayerInput
{
    readonly RandomPlayer _p = new(salt, seed);
    public GameLog Log => _p.Log;
    public ValueTask<IReadOnlyList<DraftPick>> DraftAsync(ArmyReadModel a, int r, CancellationToken ct) => r == 0 ? ValueTask.FromResult<IReadOnlyList<DraftPick>>([]) : _p.DraftAsync(a, r, ct);
    public ValueTask<IReadOnlyList<DraftPick>> DraftAsync(ArmyReadModel a, UnitOffer o, int r, CancellationToken ct) => r == 0 ? ValueTask.FromResult<IReadOnlyList<DraftPick>>([]) : _p.DraftAsync(a, o, r, ct);
    public ValueTask<PlayerChoice> ChooseAsync(ArmyReadModel a, GeneralOutput g, int r, CancellationToken ct) => r == 0 ? ValueTask.FromResult(new PlayerChoice(DoctrineSet.Empty, true)) : _p.ChooseAsync(a, g, r, ct);
    public ValueTask<PlayerChoice> ChooseAsync(ArmyReadModel a, GeneralOutput g, OrderOffer o, int r, CancellationToken ct) => r == 0 ? ValueTask.FromResult(new PlayerChoice(DoctrineSet.Empty, true)) : _p.ChooseAsync(a, g, o, r, ct);
    public ValueTask RoundClosedAsync(RoundRecord rec, ArmyReadModel after, CancellationToken ct) => _p.RoundClosedAsync(rec, after, ct);
}
